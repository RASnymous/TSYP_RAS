==============================================================================
  A. ONE-TIME SETUP
==============================================================================

--- A.1 Ubuntu VM ---
# [VM terminal] refresh the package lists
sudo apt update
# [VM terminal] the ROS packages the robots use: Gazebo bridge, EKF, SLAM, URDF tools, camera conversion, keyboard driving, TF tools
sudo apt install -y ros-lyrical-ros-gz ros-lyrical-robot-localization ros-lyrical-slam-toolbox ros-lyrical-xacro ros-lyrical-robot-state-publisher ros-lyrical-joint-state-publisher ros-lyrical-cv-bridge python3-opencv ros-lyrical-teleop-twist-keyboard ros-lyrical-tf2-tools
# [VM terminal] C compiler (beacon simulator), unzip, curl, Node.js (dashboard), numpy (ONA, simulators), serial ports, plots
sudo apt install -y build-essential unzip curl nodejs npm python3-numpy python3-serial python3-matplotlib
# [VM terminal] every new terminal loads ROS 2 (adds the line only if it is missing)
grep -q 'opt/ros/lyrical' ~/.bashrc || echo 'source /opt/ros/lyrical/setup.bash' >> ~/.bashrc
# [VM terminal] lets you use USB serial ports and VirtualBox shared folders; takes effect after a reboot
sudo usermod -aG dialout,vboxsf $USER

# [Serial Monitor, 115200 baud] mission key (demo key; KEY <32 hex> for a real mission). Stored in flash.
KEY demo
# [Serial Monitor, 115200 baud] set the clock in unix seconds (beacons have no real-time clock; use the current time)
TIME 1790000000
# [Serial Monitor, 115200 baud] network id (default 42)
NET 42
# [Serial Monitor, 115200 baud] node id (a beacon also takes it from PROV)
ID 7
# [Serial Monitor, 115200 baud] gateway label shown on the dashboard
NAME gw1
# [Serial Monitor, 115200 baud] install this beacon's record: it starts announcing it within 0.8 s
PROV {"beacon_id":7,"kind":"VICTIM","gps":{"lat":36.8066,"lon":10.1816,"err_m":2.1},"severity":15,"next":{"id":6,"dist_m":8.5}}
# [Serial Monitor, 115200 baud] give it a record it should already know
SEED {"beacon_id":6,"kind":"WAYPOINT","gps":{"lat":36.8065,"lon":10.1815}}
# [Serial Monitor, 115200 baud] withdraw this beacon's event (RETRACTED flag, new seq)
RETRACT
# [Serial Monitor, 115200 baud] one JSON line: id, role, key set, clock set, records known, duty used, frames sent and received (incl. bad MACs)
STATUS
# [Serial Monitor, 115200 baud] every record in the cache
DUMP
# [Serial Monitor, 115200 baud] print every frame received and sent (VERBOSE 0 to stop)
VERBOSE 1
# [Serial Monitor, 115200 baud] gateway: raw frame lines for the ONA on (default; RAW 0 off)
RAW 1
# [Serial Monitor, 115200 baud] gateway: transmit one 44-byte frame (the ONA does it for briefings): OK TX type 0x24
TX <88 hex characters>
# [Serial Monitor, 115200 baud] restart (record and seq survive in flash)
REBOOT
# [Serial Monitor, 115200 baud] erase key, id and record
FACTORY
# [Serial Monitor, 115200 baud] list the commands
HELP
